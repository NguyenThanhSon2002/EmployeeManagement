/**

 * Bao gồm: gán ca đơn lẻ, nhân viên tự nhận ca, hủy nhận ca, gán hàng loạt, xóa ca và cập nhật giải trình vi phạm.

@Service
public class ScheduleAssignmentService {

    private static final Logger log = LoggerFactory.getLogger(ScheduleAssignmentService.class);

    public ScheduleAssignmentService(
            ScheduleRepository scheduleRepository,
            ShiftAssignmentRepository assignmentRepository,

            EmployeeRepository employeeRepository,
            EmployeeSecondaryBranchRepository secondaryBranchRepository,
            WorkShiftRepository workShiftRepository,

    public void assertScheduleNotLocked(Schedule schedule) {
        if ("PUBLISHED".equalsIgnoreCase(schedule.getStatus())) {
            throw new BusinessException(
                    "SCHEDULE_LOCKED",
                    "Lịch làm việc đã được công bố chính thức và đang bị khóa. Chỉ Giám đốc hoặc Admin mới có quyền thu hồi về bản nháp để điều chỉnh.",
                    HttpStatus.FORBIDDEN
            );
        }
    }

    @Transactional
    public ShiftAssignmentDto assignSingleShift(AssignSingleShiftRequest request, Long currentUserId) {
        Schedule schedule = scheduleRepository.findById(request.getScheduleId())
                .orElseThrow(() -> new ResourceNotFoundException("Lịch làm việc", request.getScheduleId()));

        assertScheduleNotLocked(schedule);

        if (request.getScheduleVersion() != null && !schedule.getVersion().equals(request.getScheduleVersion())) {
            throw new ObjectOptimisticLockingFailureException(Schedule.class, schedule.getId());
        }

        if (currentUserId != null && currentUserId > 1L) {
            Employee actor = employeeRepository.findByUserIdAndDeletedAtIsNull(currentUserId).orElse(null);
            if (actor != null && actor.getPositionId() != null && actor.getPositionId() == 2) { // ROLE_MANAGER
                if (!schedule.getBranchId().equals(actor.getBranchId())) {
                    throw new BusinessException("FORBIDDEN", "Quản lý không có quyền gán ca cho lịch làm việc thuộc cơ sở khác.", HttpStatus.FORBIDDEN);
                }
            }
        }

        WorkShift shift = workShiftRepository.findByIdAndDeletedAtIsNull(request.getWorkShiftId())
                .orElseThrow(() -> new ResourceNotFoundException("Mẫu ca làm việc", request.getWorkShiftId()));

        if (shift.getBranchId() != null && !shift.getBranchId().equals(schedule.getBranchId())) {
            throw new BusinessException("BRANCH_SHIFT_MISMATCH", "Mẫu ca làm việc (" + shift.getShiftName() + ") không thuộc chi nhánh của lịch làm việc.", HttpStatus.BAD_REQUEST);
        }

        Employee targetEmp = employeeRepository.findById(request.getEmployeeId())
                .orElseThrow(() -> new ResourceNotFoundException("Nhân viên", request.getEmployeeId()));
        if (targetEmp.getPositionId() != null && targetEmp.getPositionId() == 1L) {
            throw new BusinessException("DIRECTOR_CANNOT_BE_SCHEDULED", "Không thể phân ca làm việc cho nhân sự có chức danh Giám đốc.", HttpStatus.BAD_REQUEST);
        }

        ConflictValidationResult validation = conflictEngine.validateAssignment(
                request.getEmployeeId(),
                request.getWorkShiftId(),
                request.getWorkDate(),
                null,
                schedule.getId(),
                schedule.getBranchId()
        );

        if (validation.getHasMandatoryErrors()) {
            ScheduleConflictDto err = validation.getMandatoryErrors().get(0);
            throw new BusinessException(err.getRuleCode(), err.getMessage(), HttpStatus.UNPROCESSABLE_ENTITY);
        }

        String warningRule = null;
        if (validation.getHasWarnings()) {
            warningRule = validation.getWarnings().get(0).getRuleCode();
        }

        LocalDateTime[] times = ConflictDetectionEngine.calculatePlannedTimes(
                request.getWorkDate(),
                shift.getStartTime(),
                shift.getEndTime(),
                Boolean.TRUE.equals(shift.getIsCrossMidnight())
        );

        ShiftAssignment assignment = new ShiftAssignment();
        assignment.setScheduleId(schedule.getId());
        assignment.setEmployeeId(request.getEmployeeId());
        assignment.setWorkShiftId(shift.getId());
        assignment.setWorkDate(request.getWorkDate());
        assignment.setSnapshotShiftName(shift.getShiftName());
        assignment.setSnapshotStartTime(shift.getStartTime());
        assignment.setSnapshotEndTime(shift.getEndTime());
        assignment.setSnapshotBreakMinutes(shift.getBreakMinutes());
        assignment.setSnapshotCrossMidnight(shift.getIsCrossMidnight());
        assignment.setSnapshotColorCode(shift.getColorCode());
        assignment.setPlannedStartTime(times[0]);
        assignment.setPlannedEndTime(times[1]);
        assignment.setStatus("ASSIGNED");
        assignment.setOverrideRule(request.getOverrideRule() != null ? request.getOverrideRule() : warningRule);
        assignment.setOverrideReason(request.getOverrideReason());

        ShiftAssignment saved = assignmentRepository.save(assignment);

        schedule.setVersion(schedule.getVersion() + 1);
        scheduleRepository.save(schedule);

        if ("PUBLISHED".equalsIgnoreCase(schedule.getStatus())) {
            historyRepository.save(new ScheduleChangeHistory(
                    schedule.getId(),
                    saved.getId(),
                    "CREATE",
                    null,
                    "Gán ca: " + shift.getShiftName(),
                    request.getOverrideReason() != null ? request.getOverrideReason() : "Thêm ca sau công bố",
                    currentUserId != null ? currentUserId : 1L
            ));
        }

        return toDto(saved);
    }

    @Transactional
    public ShiftAssignmentDto claimShift(ClaimShiftRequest request, Long currentUserId) {
        Schedule schedule = scheduleRepository.findById(request.getScheduleId())
                .orElseThrow(() -> new ResourceNotFoundException("Lịch làm việc", request.getScheduleId()));

        Employee emp = employeeRepository.findByUserIdAndDeletedAtIsNull(currentUserId)
                .orElseThrow(() -> new BusinessException("EMPLOYEE_NOT_FOUND", "Không tìm thấy hồ sơ nhân sự của bạn trong hệ thống.", HttpStatus.NOT_FOUND));

        if (emp.getPositionId() != null && emp.getPositionId() == 1L) {
            throw new BusinessException("DIRECTOR_CANNOT_BE_SCHEDULED", "Không thể phân ca làm việc cho Giám đốc.", HttpStatus.BAD_REQUEST);
        }

        boolean allowedInBranch = schedule.getBranchId().equals(emp.getBranchId())
                || secondaryBranchRepository.existsByEmployeeIdAndBranchId(emp.getId(), schedule.getBranchId());
        if (!allowedInBranch) {
            throw new BusinessException("BRANCH_MISMATCH", "Bạn chỉ được đăng ký ca làm việc thuộc cơ sở của bạn phụ trách.", HttpStatus.FORBIDDEN);
        }

        WorkShift shift = workShiftRepository.findByIdAndDeletedAtIsNull(request.getWorkShiftId())
                .orElseThrow(() -> new ResourceNotFoundException("Mẫu ca làm việc", request.getWorkShiftId()));

        if (shift.getBranchId() != null && !shift.getBranchId().equals(schedule.getBranchId())) {
            throw new BusinessException("BRANCH_SHIFT_MISMATCH", "Mẫu ca làm việc không thuộc chi nhánh của lịch làm việc.", HttpStatus.BAD_REQUEST);
        }

        List<ShiftAssignment> existingInShift = assignmentRepository.findAssignmentsByShiftAndDate(
                schedule.getId(), shift.getId(), request.getWorkDate()
        );

        boolean alreadyAssigned = existingInShift.stream()
                .anyMatch(a -> a.getEmployeeId().equals(emp.getId()));
        if (alreadyAssigned) {
            throw new BusinessException("ALREADY_ASSIGNED", "Bạn đã được phân công vào ca làm việc này rồi.", HttpStatus.BAD_REQUEST);
        }

        int maxCapacity = shift.getMaxStaffing() != null ? shift.getMaxStaffing() : (shift.getMinStaffing() != null ? shift.getMinStaffing() : 1);
        if (existingInShift.size() >= maxCapacity) {
            throw new BusinessException("SHIFT_FULL", "Ca làm việc đã đủ định biên nhân sự tối đa (" + existingInShift.size() + "/" + maxCapacity + " NV).", HttpStatus.BAD_REQUEST);
        }

        ConflictValidationResult validation = conflictEngine.validateAssignment(
                emp.getId(),
                shift.getId(),
                request.getWorkDate(),
                null,
                schedule.getId(),
                schedule.getBranchId()
        );

        if (validation.getHasMandatoryErrors()) {
            ScheduleConflictDto err = validation.getMandatoryErrors().get(0);
            throw new BusinessException(err.getRuleCode(), err.getMessage(), HttpStatus.UNPROCESSABLE_ENTITY);
        }

        if (validation.getHasWarnings() && !Boolean.TRUE.equals(request.getSelfVolunteeredConfirm())) {
            ScheduleConflictDto warn = validation.getWarnings().get(0);
            throw new BusinessException("WARN_VOLUNTEER_CONFIRMATION_REQUIRED",
                    warn.getMessage(), HttpStatus.BAD_REQUEST);
        }

        LocalDateTime[] times = ConflictDetectionEngine.calculatePlannedTimes(
                request.getWorkDate(),
                shift.getStartTime(),
                shift.getEndTime(),
                Boolean.TRUE.equals(shift.getIsCrossMidnight())
        );

        ShiftAssignment assignment = new ShiftAssignment();
        assignment.setScheduleId(schedule.getId());
        assignment.setEmployeeId(emp.getId());
        assignment.setWorkShiftId(shift.getId());
        assignment.setWorkDate(request.getWorkDate());
        assignment.setSnapshotShiftName(shift.getShiftName());
        assignment.setSnapshotStartTime(shift.getStartTime());
        assignment.setSnapshotEndTime(shift.getEndTime());
        assignment.setSnapshotBreakMinutes(shift.getBreakMinutes());
        assignment.setSnapshotCrossMidnight(shift.getIsCrossMidnight());
        assignment.setSnapshotColorCode(shift.getColorCode());
        assignment.setPlannedStartTime(times[0]);
        assignment.setPlannedEndTime(times[1]);
        assignment.setStatus("ASSIGNED");

        String overrideRule = validation.getHasWarnings() ? validation.getWarnings().get(0).getRuleCode() : null;
        String overrideReason = Boolean.TRUE.equals(request.getSelfVolunteeredConfirm())
                ? "Nhân viên tự nguyện đăng ký nhận ca"
                : "Nhân viên đăng ký nhận ca trống";

        assignment.setOverrideRule(overrideRule);
        assignment.setOverrideReason(overrideReason);

        ShiftAssignment saved = assignmentRepository.save(assignment);

        schedule.setVersion(schedule.getVersion() + 1);
        scheduleRepository.save(schedule);

        if ("PUBLISHED".equalsIgnoreCase(schedule.getStatus())) {
            historyRepository.save(new ScheduleChangeHistory(
                    schedule.getId(),
                    saved.getId(),
                    "CREATE",
                    null,
                    "Nhân viên tự nhận ca: " + shift.getShiftName(),
                    overrideReason,
                    currentUserId != null ? currentUserId : 1L
            ));
        }

        return toDto(saved);
    }

    @Transactional
    public void cancelClaimedShift(Long assignmentId, Long currentUserId) {
        ShiftAssignment assignment = assignmentRepository.findById(assignmentId)
                .orElseThrow(() -> new ResourceNotFoundException("Phân ca", assignmentId));

        Employee emp = employeeRepository.findByUserIdAndDeletedAtIsNull(currentUserId)
                .orElseThrow(() -> new BusinessException("EMPLOYEE_NOT_FOUND", "Không tìm thấy hồ sơ nhân sự của bạn trong hệ thống.", HttpStatus.NOT_FOUND));

        if (!assignment.getEmployeeId().equals(emp.getId())) {
            throw new BusinessException("FORBIDDEN", "Bạn chỉ được phép hủy ca làm việc do chính mình đã đăng ký.", HttpStatus.FORBIDDEN);
        }

        Schedule schedule = scheduleRepository.findById(assignment.getScheduleId())
                .orElseThrow(() -> new ResourceNotFoundException("Lịch làm việc", assignment.getScheduleId()));

        LocalDateTime now = LocalDateTime.now();
        if (assignment.getPlannedStartTime() != null && assignment.getPlannedStartTime().isBefore(now)) {
            throw new BusinessException("PAST_SHIFT", "Không thể hủy ca làm việc đã diễn ra trong quá khứ.", HttpStatus.BAD_REQUEST);
        }

        assignmentRepository.delete(assignment);

        schedule.setVersion(schedule.getVersion() + 1);
        scheduleRepository.save(schedule);

        if ("PUBLISHED".equalsIgnoreCase(schedule.getStatus())) {
            historyRepository.save(new ScheduleChangeHistory(
                    schedule.getId(),
                    assignment.getId(),
                    "DELETE",
                    assignment.getSnapshotShiftName(),
                    null,
                    "Nhân viên tự hủy ca đã đăng ký",
                    currentUserId != null ? currentUserId : 1L
            ));
        }
    }

    @Transactional
    public List<ShiftAssignmentDto> bulkAssign(BulkAssignRequest request, Long currentUserId) {
        Schedule schedule = scheduleRepository.findById(request.getScheduleId())
                .orElseThrow(() -> new ResourceNotFoundException("Lịch làm việc", request.getScheduleId()));

        assertScheduleNotLocked(schedule);

        if (request.getScheduleVersion() != null && !schedule.getVersion().equals(request.getScheduleVersion())) {
            throw new ObjectOptimisticLockingFailureException(Schedule.class, schedule.getId());
        }

        WorkShift shift = workShiftRepository.findByIdAndDeletedAtIsNull(request.getWorkShiftId())
                .orElseThrow(() -> new ResourceNotFoundException("Mẫu ca làm việc", request.getWorkShiftId()));

        if (shift.getBranchId() != null && !shift.getBranchId().equals(schedule.getBranchId())) {
            throw new BusinessException("BRANCH_SHIFT_MISMATCH", "Mẫu ca làm việc (" + shift.getShiftName() + ") không thuộc chi nhánh của lịch làm việc.", HttpStatus.BAD_REQUEST);
        }

        ConflictValidationResult validation = conflictEngine.validateBulk(
                request.getEmployeeIds(),
                request.getWorkShiftId(),
                request.getFromDate(),
                request.getToDate()
        );

        if (validation.getHasMandatoryErrors()) {
            ScheduleConflictDto err = validation.getMandatoryErrors().get(0);
            throw new BusinessException(err.getRuleCode(), err.getMessage(), HttpStatus.UNPROCESSABLE_ENTITY);
        }

        List<ShiftAssignment> savedList = new ArrayList<>();
        for (Long empId : request.getEmployeeIds()) {
            LocalDate curr = request.getFromDate();
            while (!curr.isAfter(request.getToDate())) {
                LocalDateTime[] times = ConflictDetectionEngine.calculatePlannedTimes(
                        curr,
                        shift.getStartTime(),
                        shift.getEndTime(),
                        Boolean.TRUE.equals(shift.getIsCrossMidnight())
                );

                ShiftAssignment assignment = new ShiftAssignment();
                assignment.setScheduleId(schedule.getId());
                assignment.setEmployeeId(empId);
                assignment.setWorkShiftId(shift.getId());
                assignment.setWorkDate(curr);
                assignment.setSnapshotShiftName(shift.getShiftName());
                assignment.setSnapshotStartTime(shift.getStartTime());
                assignment.setSnapshotEndTime(shift.getEndTime());
                assignment.setSnapshotBreakMinutes(shift.getBreakMinutes());
                assignment.setSnapshotCrossMidnight(shift.getIsCrossMidnight());
                assignment.setSnapshotColorCode(shift.getColorCode());
                assignment.setPlannedStartTime(times[0]);
                assignment.setPlannedEndTime(times[1]);
                assignment.setStatus("ASSIGNED");

                savedList.add(assignmentRepository.save(assignment));
                curr = curr.plusDays(1);
            }
        }

        schedule.setVersion(schedule.getVersion() + 1);
        scheduleRepository.save(schedule);

        if ("PUBLISHED".equalsIgnoreCase(schedule.getStatus())) {
            for (ShiftAssignment sa : savedList) {
                historyRepository.save(new ScheduleChangeHistory(
                        schedule.getId(),
                        sa.getId(),
                        "CREATE",
                        null,
                        "Gán ca hàng loạt: " + shift.getShiftName(),
                        "Gán ca hàng loạt sau công bố",
                        currentUserId != null ? currentUserId : 1L
                ));
            }
        }

        return savedList.stream().map(this::toDto).collect(Collectors.toList());
    }

    @Transactional
    public List<ShiftAssignmentDto> batchAssignShifts(BatchAssignItemsRequest request, Long currentUserId) {
        Schedule schedule = scheduleRepository.findById(request.getScheduleId())
                .orElseThrow(() -> new ResourceNotFoundException("Lịch làm việc", request.getScheduleId()));

        assertScheduleNotLocked(schedule);

        if (request.getScheduleVersion() != null && !schedule.getVersion().equals(request.getScheduleVersion())) {
            throw new ObjectOptimisticLockingFailureException(Schedule.class, schedule.getId());
        }

        boolean isReplaceAll = request.getMode() == null || "REPLACE_ALL".equalsIgnoreCase(request.getMode());

        if (isReplaceAll) {
            LocalDate start = request.getFromDate() != null ? request.getFromDate() : schedule.getStartDate();
            LocalDate end = request.getToDate() != null ? request.getToDate() : schedule.getEndDate();
            List<ShiftAssignment> inRange = assignmentRepository.findAllByScheduleId(schedule.getId()).stream()
                    .filter(a -> a.getWorkDate() != null && !a.getWorkDate().isBefore(start) && !a.getWorkDate().isAfter(end))
                    .toList();
            if (!inRange.isEmpty()) {
                assignmentRepository.deleteAll(inRange);
            }
        }

        // Preload existing assignments for this schedule to avoid duplicates
        List<ShiftAssignment> existingList = assignmentRepository.findAllByScheduleId(schedule.getId());
        Map<String, ShiftAssignment> existingMap = new HashMap<>();
        for (ShiftAssignment sa : existingList) {
            if (!"CANCELLED".equalsIgnoreCase(sa.getStatus())) {
                String key = sa.getEmployeeId() + "_" + sa.getWorkShiftId() + "_" + sa.getWorkDate();
                existingMap.put(key, sa);
            }
        }

        // Cache work shifts
        Set<Long> shiftIds = request.getAssignments().stream()
                .map(BatchAssignItemsRequest.AssignItem::getWorkShiftId)
                .collect(Collectors.toSet());
        Map<Long, WorkShift> shiftMap = workShiftRepository.findAllById(shiftIds).stream()
                .collect(Collectors.toMap(WorkShift::getId, s -> s));

        // Cache employees to exclude Director
        Set<Long> empIds = request.getAssignments().stream()
                .map(BatchAssignItemsRequest.AssignItem::getEmployeeId)
                .collect(Collectors.toSet());
        Map<Long, Employee> empMap = employeeRepository.findAllById(empIds).stream()
                .collect(Collectors.toMap(Employee::getId, e -> e));

        List<ShiftAssignment> toSave = new ArrayList<>();
        for (BatchAssignItemsRequest.AssignItem item : request.getAssignments()) {
            Employee emp = empMap.get(item.getEmployeeId());
            if (emp != null && emp.getPositionId() != null && emp.getPositionId() == 1L) {
                // Giám đốc không tham gia trực ca
                continue;
            }
            WorkShift shift = shiftMap.get(item.getWorkShiftId());
            if (shift == null) {
                shift = workShiftRepository.findByIdAndDeletedAtIsNull(item.getWorkShiftId())
                        .orElseThrow(() -> new ResourceNotFoundException("Mẫu ca làm việc", item.getWorkShiftId()));
                shiftMap.put(shift.getId(), shift);
            }

            if (shift.getBranchId() != null && !shift.getBranchId().equals(schedule.getBranchId())) {
                throw new BusinessException("BRANCH_SHIFT_MISMATCH", "Mẫu ca làm việc (" + shift.getShiftName() + ") không thuộc chi nhánh của lịch làm việc.", HttpStatus.BAD_REQUEST);
            }

            String key = item.getEmployeeId() + "_" + item.getWorkShiftId() + "_" + item.getWorkDate();
            ShiftAssignment target = existingMap.get(key);

            LocalDateTime[] times = ConflictDetectionEngine.calculatePlannedTimes(
                    item.getWorkDate(),
                    shift.getStartTime(),
                    shift.getEndTime(),
                    Boolean.TRUE.equals(shift.getIsCrossMidnight())
            );

            if (target != null) {
                // Update existing assignment
                target.setSnapshotShiftName(shift.getShiftName());
                target.setSnapshotStartTime(shift.getStartTime());
                target.setSnapshotEndTime(shift.getEndTime());
                target.setSnapshotBreakMinutes(shift.getBreakMinutes());
                target.setSnapshotCrossMidnight(shift.getIsCrossMidnight());
                target.setSnapshotColorCode(shift.getColorCode());
                target.setPlannedStartTime(times[0]);
                target.setPlannedEndTime(times[1]);
                target.setStatus("ASSIGNED");
                target.setOverrideRule(item.getOverrideRule());
                target.setOverrideReason(item.getOverrideReason());
                toSave.add(target);
            } else {
                // Create new assignment
                ShiftAssignment assignment = new ShiftAssignment();
                assignment.setScheduleId(schedule.getId());
                assignment.setEmployeeId(item.getEmployeeId());
                assignment.setWorkShiftId(shift.getId());
                assignment.setWorkDate(item.getWorkDate());
                assignment.setSnapshotShiftName(shift.getShiftName());
                assignment.setSnapshotStartTime(shift.getStartTime());
                assignment.setSnapshotEndTime(shift.getEndTime());
                assignment.setSnapshotBreakMinutes(shift.getBreakMinutes());
                assignment.setSnapshotCrossMidnight(shift.getIsCrossMidnight());
                assignment.setSnapshotColorCode(shift.getColorCode());
                assignment.setPlannedStartTime(times[0]);
                assignment.setPlannedEndTime(times[1]);
                assignment.setStatus("ASSIGNED");
                assignment.setOverrideRule(item.getOverrideRule());
                assignment.setOverrideReason(item.getOverrideReason());
                toSave.add(assignment);
                existingMap.put(key, assignment);
            }
        }

        List<ShiftAssignment> saved = assignmentRepository.saveAll(toSave);

        schedule.setVersion(schedule.getVersion() + 1);
        scheduleRepository.save(schedule);

        if ("PUBLISHED".equalsIgnoreCase(schedule.getStatus())) {
            for (ShiftAssignment sa : saved) {
                historyRepository.save(new ScheduleChangeHistory(
                        schedule.getId(),
                        sa.getId(),
                        "CREATE",
                        null,
                        "Gán phân ca tự động hàng loạt: " + sa.getSnapshotShiftName(),
                        "Xếp ca tự động",
                        currentUserId != null ? currentUserId : 1L
                ));
            }
        }

        return saved.stream().map(this::toDto).collect(Collectors.toList());
    }

            log.warn("Schedule version mismatch on override update for assignment {}: client={}, server={}",
                    id, request.getScheduleVersion(), schedule.getVersion());
        }

        // Kiểm tra quyền hạn quản lý: Quản lý không được sửa ca của cơ sở khác

                }
            }
        }

        String rawReason = request.getOverrideReason();
        String cleanedReason = (rawReason != null && !rawReason.trim().isEmpty()) ? rawReason.trim() : null;
        assignment.setOverrideReason(cleanedReason);

        if (request.getOverrideRule() != null) {
            assignment.setOverrideRule(request.getOverrideRule().trim().isEmpty() ? null : request.getOverrideRule().trim());
        }

                    saved.getId(),
                    "UPDATE_OVERRIDE",
                    null,
                    cleanedReason != null ? ("Cập nhật giải trình ca: " + assignment.getSnapshotShiftName()) : ("Xóa giải trình ca: " + assignment.getSnapshotShiftName()),
                    cleanedReason,
                    currentUserId != null ? currentUserId : 1L
            ));
        }

        schedule.setVersion(schedule.getVersion() + 1);
        scheduleRepository.save(schedule);

        return toDto(saved);
    }

@Component
public class StaffingCapacityRule implements ConflictRule {

    @Override
    public void evaluate(ConflictDetectionContext context, ConflictValidationResult result) {
        WorkShift shift = context.getWorkShift();
        if (shift == null || shift.getMinStaffing() == null || shift.getMaxStaffing() == null) {
            return;
        }

        // Count assignments on the same workDate for this shift
        int count = 1; // including the candidate
        for (ShiftAssignment existing : context.getExistingAssignments()) {
            if ("CANCELLED".equalsIgnoreCase(existing.getStatus())) {
                continue;
            }
            if (context.getExcludeAssignmentId() != null && context.getExcludeAssignmentId().equals(existing.getId())) {
                continue;
            }
            if (context.getWorkDate().equals(existing.getWorkDate()) &&
                    context.getWorkShiftId().equals(existing.getWorkShiftId()) &&
                    !context.getEmployeeId().equals(existing.getEmployeeId())) {
                count++;
            }
        }

        String empName = context.getEmployee() != null ? context.getEmployee().getFullName() : "Nhân viên #" + context.getEmployeeId();

        if (count > shift.getMaxStaffing()) {
            String msg = String.format("%s ngày %s đã vượt định biên tối đa (%d NV). Bổ sung nhân sự cần ghi nhận lý do tăng cường.",
                    shift.getShiftName(), context.getWorkDate(), shift.getMaxStaffing());

            result.addWarning(new ScheduleConflictDto(
                    "WARN_STAFFING_CAPACITY",
                    "WARNING",
                    context.getEmployeeId(),
                    empName,
                    context.getWorkDate(),
                    context.getWorkShiftId(),
                    msg
            ));
        }
    }
}

package com.company.hrms.modules.schedule.engine.rule;

import com.company.hrms.modules.schedule.domain.ShiftAssignment;
import com.company.hrms.modules.schedule.dto.ConflictValidationResult;
import com.company.hrms.modules.schedule.dto.ScheduleConflictDto;
import com.company.hrms.modules.schedule.engine.ConflictDetectionContext;
import com.company.hrms.modules.schedule.engine.rule.IntervalMergeUtils.Interval;
import static com.company.hrms.modules.schedule.engine.rule.IntervalMergeUtils.calculateDeduplicatedMinutes;

import org.springframework.stereotype.Component;

import java.time.DayOfWeek;
import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;

@Component
public class WeeklyMaxHoursRule implements ConflictRule {

    @Override
    public void evaluate(ConflictDetectionContext context, ConflictValidationResult result) {
        if (context.getPlannedStartTime() == null || context.getPlannedEndTime() == null) {
            return;
        }

        LocalDate monday = context.getWorkDate().with(DayOfWeek.MONDAY);
        LocalDate sunday = context.getWorkDate().with(DayOfWeek.SUNDAY);

        List<Interval> intervals = new ArrayList<>();

        // 1. Candidate shift hours
        int candidateBreak = context.getWorkShift() != null && context.getWorkShift().getBreakMinutes() != null
                ? context.getWorkShift().getBreakMinutes()
                : 0;
        intervals.add(new Interval(context.getPlannedStartTime(), context.getPlannedEndTime(), candidateBreak));

        // 2. Existing assignments in this week for this employee
        for (ShiftAssignment existing : context.getExistingAssignments()) {
            if ("CANCELLED".equalsIgnoreCase(existing.getStatus())) {
                continue;
            }
            if (context.getExcludeAssignmentId() != null && context.getExcludeAssignmentId().equals(existing.getId())) {
                continue;
            }
            if (!context.getEmployeeId().equals(existing.getEmployeeId())) {
                continue;
            }

            LocalDate existingDate = existing.getWorkDate();
            if (!existingDate.isBefore(monday) && !existingDate.isAfter(sunday)) {
                int existBreak = existing.getSnapshotBreakMinutes() != null ? existing.getSnapshotBreakMinutes() : 0;
                intervals.add(new Interval(existing.getPlannedStartTime(), existing.getPlannedEndTime(), existBreak));
            }
        }

        long totalMinutes = calculateDeduplicatedMinutes(intervals);
        double totalHours = totalMinutes / 60.0;
        if (totalHours > context.getMaxWeeklyHours()) {
            String empName = context.getEmployee() != null ? context.getEmployee().getFullName() : "Nhân viên #" + context.getEmployeeId();
            String msg = String.format("Tổng giờ làm trong tuần (từ %s đến %s) của nhân viên %s là %.1f giờ (vượt giới hạn quy định %.1f giờ/tuần).",
                    monday, sunday, empName, totalHours, context.getMaxWeeklyHours());

            result.addWarning(new ScheduleConflictDto(
                    "WARN_WEEKLY_HOURS_EXCEEDED",
                    "WARNING",
                    context.getEmployeeId(),
                    empName,
                    context.getWorkDate(),
                    context.getWorkShiftId(),
                    msg
            ));
        }
    }
}

  const today = new Date();

  const diffToMonday = dayOfWeek === 0 ? -6 : 1 - dayOfWeek;

  const currentMondayStr = format(monday, 'yyyy-MM-dd');

  // Attempt initial load from localStorage

  return {

    setFilters: (newFilters) => {

    assignShift: (payload) => {
      const state = get();
      if (state.scheduleStatus === 'PUBLISHED') {
        return {
          success: false,
          error: 'Lịch tuần đã công bố chính thức và đang bị khóa. Chỉ Giám đốc hoặc Admin mới có quyền thu hồi về bản nháp để điều chỉnh.',
        };
      }

      const emp = state.employees.find((e) => Number(e.id) === Number(payload.employeeId));
      const template = state.shiftTemplates.find((t) => Number(t.id) === Number(payload.workShiftId));

      if (!emp || !template) {
        return { success: false, error: 'Nhân viên hoặc Ca làm việc không tồn tại' };
      }

      if (!emp.isActive || emp.employmentStatus === 'RESIGNED') {
        return {
          success: false,
          error: 'Nhân viên đã thôi việc hoặc đang bị vô hiệu hóa, không được phân ca.',
        };
      }

      const key = `${payload.employeeId}_${payload.workDate}`;
      const existingList = state.assignments[key] ? [...state.assignments[key]] : [];

      const { startMs, endMs } = getShiftAbsoluteTimeRange(
          payload.workDate,
          template.startTime,
          template.endTime,
          template.isCrossMidnight
      );

      // Determine if updating an existing assignment or adding a new one
      const existingIndex = payload.assignmentId !== undefined
          ? existingList.findIndex((a) => a.id === payload.assignmentId)
          : -1;

      const existingAssignment = existingIndex >= 0 ? existingList[existingIndex] : null;

      const newAssignment: ShiftAssignment = {
        id: existingAssignment ? existingAssignment.id : (Date.now() + Math.floor(Math.random() * 10000)),
        scheduleId: state.scheduleId,
        employeeId: emp.id,
        workShiftId: template.id,
        workDate: payload.workDate,
        snapshotShiftName: template.shiftName,
        snapshotStartTime: template.startTime,
        snapshotEndTime: template.endTime,
        snapshotBreakMinutes: template.breakMinutes,
        snapshotCrossMidnight: template.isCrossMidnight,
        snapshotColorCode: template.colorCode,
        plannedStartTime: new Date(startMs).toISOString(),
        plannedEndTime: new Date(endMs).toISOString(),
        status: 'ASSIGNED',
        overrideRule: payload.overrideRule ?? null,
        overrideReason: payload.overrideReason ?? null,
        version: existingAssignment ? existingAssignment.version + 1 : 1,
      };

      if (existingIndex >= 0) {
        existingList[existingIndex] = newAssignment;
      } else {
        existingList.push(newAssignment);
      }

      const updatedAssignments = {
        ...state.assignments,
        [key]: existingList,
      };

      // Persist to localStorage immediately (offline-first)
      saveBranchScheduleToStorage(state.filterOptions.branchId || 1, updatedAssignments, state.currentStartDate);

      const updatedConflicts = computeAllConflicts(
          updatedAssignments,
          state.employees,
          state.approvedLeaves,
          state.holidays,
          state.shiftTemplates
      );

      set({
        assignments: updatedAssignments,
        conflicts: updatedConflicts,
        changeHistory: state.changeHistory,
      });

      scheduleService
          .assignSingleShift({
            schedule_id: state.scheduleId,
            employee_id: payload.employeeId,
            work_shift_id: payload.workShiftId,
            work_date: payload.workDate,
            override_rule: payload.overrideRule || undefined,
            override_reason: payload.overrideReason || undefined,
            schedule_version: state.scheduleVersion,
          })
          .then((saved) => {
            if (saved && saved.id) {
              set((prev) => {
                const currentList = prev.assignments[key] || [];
                const updatedList = currentList.map((item) =>
                    item.id === newAssignment.id ? { ...item, id: saved.id, version: saved.version || item.version } : item
                );
                const newAssignmentsMap = {
                  ...prev.assignments,
                  [key]: updatedList,
                };
                saveBranchScheduleToStorage(prev.filterOptions.branchId || 1, newAssignmentsMap, prev.currentStartDate);
                const recomputedConflicts = computeAllConflicts(
                  newAssignmentsMap,
                  prev.employees,
                  prev.approvedLeaves,
                  prev.holidays,
                  prev.shiftTemplates
                );
                return {
                  assignments: newAssignmentsMap,
                  conflicts: recomputedConflicts,
                  scheduleVersion: (prev.scheduleVersion || 1) + 1,
                };
              });
            }
          })
          .catch((err) => {
            console.warn('Failed to persist shift assignment to backend:', err);
          });

      return { success: true };
    },

    deleteAssignment: (assignmentId, _changeReason, _actorRole = 'ROLE_DIRECTOR') => {
      const state = get();
      if (state.scheduleStatus === 'PUBLISHED') {
        console.warn('Cannot delete shift assignment on a PUBLISHED schedule without reverting to DRAFT.');
        return;
      }

      let targetKey: string | null = null;
      let targetAssignment: ShiftAssignment | null = null;

      for (const [key, list] of Object.entries(state.assignments)) {
        const found = list.find((a) => a.id === assignmentId);
        if (found) {
          targetKey = key;
          targetAssignment = found;
          break;
        }
      }

      if (!targetKey || !targetAssignment) return;

      // Bảo vệ: Không cho phép xóa ca làm việc ngoại vi (chỉ đọc)
      if (targetAssignment.readOnly || targetAssignment.isExternal) {
        console.warn('Cannot delete external read-only assignment:', targetAssignment);
        return;
      }

      scheduleService.deleteAssignment(assignmentId, state.scheduleVersion).catch((err) => {
        console.warn('Failed to delete assignment on backend:', err);
      });

      const updatedAssignments = { ...state.assignments };
      const filteredList = (updatedAssignments[targetKey] || []).filter((a) => a.id !== assignmentId);
      if (filteredList.length > 0) {
        updatedAssignments[targetKey] = filteredList;
      } else {
        delete updatedAssignments[targetKey];
      }

      saveBranchScheduleToStorage(state.filterOptions.branchId || 1, updatedAssignments, state.currentStartDate);

      const updatedConflicts = computeAllConflicts(
          updatedAssignments,
          state.employees,
          state.approvedLeaves,
          state.holidays,
          state.shiftTemplates
      );

      set({
        assignments: updatedAssignments,
        conflicts: updatedConflicts,
        changeHistory: state.changeHistory,
      });
    },

    updateAssignmentOverride: async (assignmentId: number, overrideReason: string | null, overrideRule?: string) => {
      const state = get();
      if (state.scheduleStatus === 'PUBLISHED') {
        return false;
      }

      const normalizedReason = overrideReason && overrideReason.trim().length > 0 ? overrideReason.trim() : null;

      let found = false;
      let targetKey: string | null = null;
      const updatedAssignments: Record<string, ShiftAssignment[]> = { ...state.assignments };

      for (const key of Object.keys(updatedAssignments)) {
        const list = updatedAssignments[key];
        const idx = list.findIndex((a) => Number(a.id) === Number(assignmentId));
        if (idx >= 0) {
          const updatedItem: ShiftAssignment = {
            ...list[idx],
            overrideReason: normalizedReason,
            overrideRule: normalizedReason ? (overrideRule || list[idx].overrideRule || 'WARN_STAFFING_CAPACITY') : null,
          };
          updatedAssignments[key] = [
            ...list.slice(0, idx),
            updatedItem,
            ...list.slice(idx + 1),
          ];
          found = true;
          targetKey = key;
          break;
        }
      }

      if (!found) return false;

      saveBranchScheduleToStorage(state.filterOptions.branchId || 1, updatedAssignments, state.currentStartDate);
      const updatedConflicts = computeAllConflicts(
        updatedAssignments,
        state.employees,
        state.approvedLeaves,
        state.holidays,
        state.shiftTemplates
      );

      set({
        assignments: updatedAssignments,
        conflicts: updatedConflicts,
      });

      try {
        const saved = await scheduleService.updateAssignmentOverride(assignmentId, {
          override_reason: normalizedReason,
          override_rule: normalizedReason ? overrideRule : null,
          schedule_version: state.scheduleVersion,
        });

        if (saved && targetKey) {
          set((prev) => {
            const currentList = prev.assignments[targetKey!] || [];
            const nextList = currentList.map((item) =>
              Number(item.id) === Number(assignmentId)
                ? {
                    ...item,
                    overrideReason: saved.override_reason !== undefined ? saved.override_reason : normalizedReason,
                    overrideRule: saved.override_rule !== undefined ? saved.override_rule : (normalizedReason ? item.overrideRule : null),
                    version: saved.version || item.version,
                  }
                : item
            );
            const nextMap = {
              ...prev.assignments,
              [targetKey!]: nextList,
            };
            saveBranchScheduleToStorage(prev.filterOptions.branchId || 1, nextMap, prev.currentStartDate);
            return {
              assignments: nextMap,
              scheduleVersion: (prev.scheduleVersion || 1) + 1,
            };
          });
        }
      } catch (err) {
        console.error('Failed to update override reason to backend:', err);
      }

      return true;
    },

    applyAutoAssignments: async (
        newAssignments,
        mode: 'REPLACE_ALL' | 'FILL_MISSING' = 'REPLACE_ALL',
        fromDate?: string,
        toDate?: string
    ): Promise<{ success: boolean; error?: string }> => {
      if (newAssignments.length === 0) return { success: true };
      const state = get();
      const branchId = state.filterOptions.branchId || 1;

      // 1. Cập nhật optimistic lên UI để hiển thị phản hồi ngay lập tức
      const updated: Record<string, ShiftAssignment[]> = {};

      if (mode === 'REPLACE_ALL') {
        const minDate = fromDate || newAssignments.reduce((min, a) => (a.workDate < min ? a.workDate : min), newAssignments[0].workDate);
        const maxDate = toDate || newAssignments.reduce((max, a) => (a.workDate > max ? a.workDate : max), newAssignments[0].workDate);

        for (const [key, list] of Object.entries(state.assignments)) {
          const preserved = list.filter((a) => a.workDate < minDate || a.workDate > maxDate);
          if (preserved.length > 0) {
            updated[key] = preserved;
          }
        }
      } else {
        for (const [key, list] of Object.entries(state.assignments)) {
          updated[key] = [...list];
        }
      }

      for (const a of newAssignments) {
        const key = `${a.employeeId}_${a.workDate}`;
        const list = updated[key] ? [...updated[key]] : [];
        const alreadyHasShift = list.some(
            (x) => x.workShiftId === a.workShiftId && x.status !== 'CANCELLED'
        );
        if (alreadyHasShift) continue;

        list.push(a);
        updated[key] = list;
      }

      const updatedConflicts = computeAllConflicts(
          updated,
          state.employees,
          state.approvedLeaves,
          state.holidays,
          state.shiftTemplates
      );

      set({ assignments: updated, conflicts: updatedConflicts });

      // 2. Đảm bảo có scheduleId hợp lệ từ backend trước khi lưu hàng loạt
      let targetScheduleId = state.scheduleId;
      let targetScheduleVersion = state.scheduleVersion;

      if (!targetScheduleId) {
        try {
          const scheduleData = await scheduleService.getWeeklySchedule(branchId, state.currentStartDate);
          if (scheduleData) {
            targetScheduleId = scheduleData.schedule_id;
            targetScheduleVersion = scheduleData.version !== undefined ? scheduleData.version : 0;
          }
        } catch (e) {
          console.warn('Could not fetch schedule before batch persist:', e);
        }
      }

      if (targetScheduleId) {
        try {
          await scheduleService.batchAssignShifts({
            schedule_id: targetScheduleId,
            schedule_version: targetScheduleVersion,
            mode,
            from_date: fromDate,
            to_date: toDate,
            assignments: newAssignments.map((a) => ({
              employee_id: a.employeeId,
              work_shift_id: a.workShiftId,
              work_date: a.workDate,
              override_rule: a.overrideRule || undefined,
              override_reason: a.overrideReason || undefined,
            })),
          });

          // Tải lại lịch từ máy chủ để nạp toàn bộ ID thực của MySQL vào store & storage
          await get().fetchWeeklySchedule(branchId, state.currentStartDate);
          return { success: true };
        } catch (err: any) {
          console.error('Failed to persist batch assignments to backend:', err);
          // Rollback bằng cách nạp lại trạng thái máy chủ
          await get().fetchWeeklySchedule(branchId, state.currentStartDate);
          const msg = err?.response?.data?.error?.message || err?.message || 'Không thể lưu phân ca lên máy chủ';
          return { success: false, error: msg };
        }
      }

      return { success: false, error: 'Không tìm thấy thông tin lịch làm việc để lưu.' };
    },

    resetCurrentWeekSchedule: async () => {
      const state = get();
      const branchId = state.filterOptions.branchId || 1;
      const currentMon = parseISO(state.currentStartDate);
      const currentSun = addDays(currentMon, 6);
      const currentMonStr = state.currentStartDate;
      const currentSunStr = format(currentSun, 'yyyy-MM-dd');

      // 1. Lọc bỏ toàn bộ phân ca trong khoảng thời gian tuần hiện tại
      const updatedAssignments = { ...state.assignments };
      for (const [key, list] of Object.entries(updatedAssignments)) {
        const remaining = list.filter(
            (a) => a.workDate < currentMonStr || a.workDate > currentSunStr
        );
        if (remaining.length > 0) {
          updatedAssignments[key] = remaining;
        } else {
          delete updatedAssignments[key];
        }
      }

      // 2. Lưu lại vào cache trình duyệt
      saveBranchScheduleToStorage(branchId, updatedAssignments, currentMonStr);

      // 3. Tính toán lại xung đột (conflicts sẽ trở về rỗng cho tuần này)
      const updatedConflicts = computeAllConflicts(
          updatedAssignments,
          state.employees,
          state.approvedLeaves,
          state.holidays,
          state.shiftTemplates
      );

      // 4. Đồng bộ State
      set({
        assignments: updatedAssignments,
        conflicts: updatedConflicts,
      });

      // 5. Nếu có scheduleId thực tế, gọi API dọn dẹp các phân ca trên server
      if (state.scheduleId) {
        try {
          await scheduleService.deleteAssignment(state.scheduleId, state.scheduleVersion);
        } catch (err) {
          console.warn('Lỗi khi xóa phân ca tuần trên máy chủ:', err);
        }
      }

      return true;
    },

      // Resolve branchName from directory store if not present

      set({
        shiftTemplates: state.shiftTemplates.map((t) => (t.id === id ? { ...t, ...mergedData } : t)),

          const updated = await workShiftService.updateWorkShift(id, data);

            shiftTemplates: get().shiftTemplates.map((t) => (t.id === id ? updated : t)),
          });

          set({ shiftTemplates: prevTemplates });
          throw err;

        }
      }
      return { success: true };
    },

    addApprovedLeave: (leave) => {
      const state = get();
      const existing = state.approvedLeaves.filter((l) => l.id !== leave.id);
      set({ approvedLeaves: [...existing, leave] });
      get().recomputeAllConflicts();
    },

    removeApprovedLeave: (leaveId) => {
      const state = get();
      set({ approvedLeaves: state.approvedLeaves.filter((l) => l.id !== leaveId) });
      get().recomputeAllConflicts();
    },
  };
});
